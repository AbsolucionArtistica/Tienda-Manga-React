REEMPLAZAR la función crearOrden con esto:

```javascript
const { productosData } = require('../data/productos');

exports.crearOrden = async (req, res) => {
  try {
    const { items, direccion } = req.body;
    const usuarioId = req.user.id;

    // Validar items
    if (!items || items.length === 0) {
      return res.status(400).json({
        success: false,
        mensajes: ["items debe contener al menos 1 producto"]
      });
    }

    // Validar dirección
    if (!direccion || !direccion.calle || !direccion.ciudad || !direccion.codigoPostal || !direccion.pais) {
      return res.status(400).json({
        success: false,
        mensajes: ["Todos los campos de dirección son requeridos"]
      });
    }

    // Validar y calcular total
    let total = 0;
    const itemsValidados = items.map(item => {
      // Buscar el producto en los datos locales
      const producto = productosData.find(p => p.id === parseInt(item.productoId));
      
      if (!producto) {
        throw new Error(`Producto con ID ${item.productoId} no encontrado`);
      }

      if (item.cantidad <= 0) {
        throw new Error(`Cantidad debe ser mayor a 0 para producto ${producto.nombre}`);
      }

      if (item.cantidad > producto.stock) {
        throw new Error(`Stock insuficiente para ${producto.nombre}. Disponibles: ${producto.stock}`);
      }

      const subtotal = item.cantidad * (item.precio || producto.precio);
      total += subtotal;

      return {
        productoId: item.productoId,
        nombre: item.nombre || producto.nombre,
        cantidad: item.cantidad,
        precio: item.precio || producto.precio,
        subtotal: subtotal,
        autor: item.autor || producto.autor,
        editorial: item.editorial || producto.editorial
      };
    });

    // Crear la orden (adaptarse según tu BD)
    // Ejemplo con Sequelize:
    const orden = await Orden.create({
      usuarioId: usuarioId,
      items: itemsValidados,
      total: total,
      estado: 'pendiente',
      direccion: direccion,
      createdAt: new Date(),
      updatedAt: new Date()
    });

    // O si usas MongoDB:
    // const orden = new Orden({
    //   usuarioId: usuarioId,
    //   items: itemsValidados,
    //   total: total,
    //   estado: 'pendiente',
    //   direccion: direccion
    // });
    // await orden.save();

    res.status(201).json({
      success: true,
      mensaje: "Orden creada exitosamente",
      orden: {
        id: orden.id,
        usuarioId: usuarioId,
        items: itemsValidados,
        total: total,
        estado: 'pendiente',
        direccion: direccion,
        createdAt: orden.createdAt,
        updatedAt: orden.updatedAt
      }
    });

  } catch (error) {
    console.error('Error creando orden:', error);
    res.status(400).json({
      success: false,
      mensaje: error.message || 'Error al crear la orden'
    });
  }
};
```

```javascript
exports.obtenerOrdenes = async (req, res) => {
  try {
    const usuarioId = req.user.id;

    // Sequelize
    const ordenes = await Orden.findAll({
      where: { usuarioId: usuarioId },
      order: [['createdAt', 'DESC']]
    });

    // O MongoDB
    // const ordenes = await Orden.find({ usuarioId: usuarioId }).sort({ createdAt: -1 });

    res.json({
      success: true,
      ordenes: ordenes
    });

  } catch (error) {
    console.error('Error obteniendo órdenes:', error);
    res.status(500).json({
      success: false,
      mensaje: 'Error al obtener órdenes'
    });
  }
};

exports.obtenerOrdenPorId = async (req, res) => {
  try {
    const { id } = req.params;
    const usuarioId = req.user.id;

    // Sequelize
    const orden = await Orden.findOne({
      where: { id: id, usuarioId: usuarioId }
    });

    // O MongoDB
    // const orden = await Orden.findById(id);

    if (!orden) {
      return res.status(404).json({
        success: false,
        mensaje: 'Orden no encontrada'
      });
    }

    res.json({
      success: true,
      orden: orden
    });

  } catch (error) {
    console.error('Error obteniendo orden:', error);
    res.status(500).json({
      success: false,
      mensaje: 'Error al obtener la orden'
    });
  }
};
```

```javascript
module.exports = router;
```

MODELO DE ORDEN
================================================================================

Si usas Sequelize, crea un modelo similar a esto:

```javascript
module.exports = (sequelize, DataTypes) => {
  const Orden = sequelize.define('Orden', {
    id: {
      type: DataTypes.INTEGER,
      primaryKey: true,
      autoIncrement: true
    },
    usuarioId: {
      type: DataTypes.INTEGER,
      allowNull: false
    },
    items: {
      type: DataTypes.JSON,
      allowNull: false
    },
    total: {
      type: DataTypes.FLOAT,
      allowNull: false
    },
    estado: {
      type: DataTypes.STRING,
      defaultValue: 'pendiente'
    },
    direccion: {
      type: DataTypes.JSON,
      allowNull: false
    }
  });

  return Orden;
};
```
